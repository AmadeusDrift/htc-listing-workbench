## 四 共用契约和验收规则

### 运行证据契约

新增 `src/shared/run-evidence-contracts.ts`，下列是目标公共类型，具体持久化读取由S01负责，不让每个模块各造一套状态名：

```ts
export type RequestCertainty =
  | 'NOT_SENT' | 'SENT_UNKNOWN' | 'RECEIVED_INVALID'
  | 'RECEIVED_VALID' | 'DEFINITIVE_REJECTION';
export type AbortSource =
  | 'USER' | 'TOTAL_TIMEOUT' | 'BODY_TIMEOUT' | 'SHUTDOWN' | 'NONE';
export type RunEvidenceEvent =
  | 'STAGE_STARTED' | 'REQUEST_RESERVED' | 'DISPATCH_ATTEMPTED'
  | 'HEADERS_RECEIVED' | 'BODY_COMPLETED' | 'RECEIPT_SAVED'
  | 'OUTPUT_REJECTED' | 'RESULT_STORED' | 'RESULT_ADOPTED'
  | 'STAGE_BLOCKED' | 'STAGE_COMPLETED' | 'RECOVERY_LINKED';
export interface RunEvidence {
  version: 1;
  at: string;
  event: RunEvidenceEvent;
  stage: string;
  taskId?: string;
  itemKey?: string;
  sku?: string;
  operationId?: string;
  callId?: string;
  receiptId?: string;
  draftId?: string;
  slot?: number;
  certainty?: RequestCertainty;
  abortSource?: AbortSource;
  errorCode?: string;
  httpStatus?: number;
  receivedBytes?: number;
}
```

`DISPATCH_ATTEMPTED`只说明应用进入发送，不声称服务商必已接收；从这个点开始不能凭本地断线当成未发送。`RECEIVED_INVALID`不是未知，也不自动等于允许再收费。`RECEIVED_VALID`不是已采用，`RESULT_ADOPTED`不是平台已接受或可售。当前无字段的旧记录按缺失处理，不补造时间/身份。

允许通过现有审计存储追加结构化事件；如果需要新表，必须指定单一所有者、迁移、备份和恢复语义。实时统计从事件和现有业务记录只读投影，查询不触发修复、迁移或请求。不能每次轮询追加相同日志。默认只记录安全字段，禁止Key、Authorization、完整请求头、任意响应正文、私有图片和完整商品原始资料进入通用日志。
